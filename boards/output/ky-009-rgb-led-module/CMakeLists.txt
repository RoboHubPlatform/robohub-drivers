cmake_minimum_required(VERSION 3.16)
project(robohub-board-ky-009-rgb-led-module C)

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/deps.cmake")
  include("${CMAKE_CURRENT_SOURCE_DIR}/deps.cmake")
endif()

function(robohub_locate_module module out_var)
  string(TOUPPER "${module}" upper)
  string(REPLACE "-" "_" upper "${upper}")
  if(${out_var})
    return()
  endif()
  if(DEFINED ENV{${upper}_DIR})
    set(${out_var} "$ENV{${upper}_DIR}" PARENT_SCOPE)
    return()
  endif()
  set(dir "${CMAKE_CURRENT_SOURCE_DIR}")
  foreach(level RANGE 8)
    get_filename_component(parent "${dir}/.." ABSOLUTE)
    if(EXISTS "${parent}/${module}/CMakeLists.txt")
      set(${out_var} "${parent}/${module}" PARENT_SCOPE)
      return()
    endif()
    set(dir "${parent}")
  endforeach()
  include(FetchContent)
  message(STATUS "[${PROJECT_NAME}] fetching ${module}")
  FetchContent_Declare(${module}
    GIT_REPOSITORY "https://github.com/RoboHubPlatform/${module}.git"
    GIT_TAG main GIT_SHALLOW TRUE GIT_PROGRESS TRUE)
  FetchContent_MakeAvailable(${module})
  set(${out_var} "${${module}_SOURCE_DIR}" PARENT_SCOPE)
endfunction()

robohub_locate_module(robohub-common ROBOHUB_COMMON_DIR)
robohub_locate_module(robohub-driver-rgb-pwm ROBOHUB_DRIVER_RGB_PWM_DIR)
if(NOT ROBOHUB_COMMON_DIR OR NOT ROBOHUB_DRIVER_RGB_PWM_DIR)
  message(FATAL_ERROR "robohub-common or robohub-driver-rgb-pwm not found")
endif()

if("${CMAKE_SOURCE_DIR}" STREQUAL "${CMAKE_CURRENT_SOURCE_DIR}")
  if(NOT TARGET robohub-common)
    add_subdirectory("${ROBOHUB_COMMON_DIR}" robohub-common)
  endif()
  if(NOT TARGET robohub-driver-rgb-pwm)
    add_subdirectory("${ROBOHUB_DRIVER_RGB_PWM_DIR}" robohub-driver-rgb-pwm)
  endif()
endif()

add_library(robohub-board-ky-009-rgb-led-module STATIC bsp/bsp.c)
target_include_directories(robohub-board-ky-009-rgb-led-module PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}/bsp
    ${ROBOHUB_COMMON_DIR}/include)
target_link_libraries(robohub-board-ky-009-rgb-led-module PUBLIC robohub-driver-rgb-pwm robohub-common)
target_compile_features(robohub-board-ky-009-rgb-led-module PUBLIC c_std_11)

install(FILES module.yml Kconfig DESTINATION share/robohub-board-ky-009-rgb-led-module)
install(DIRECTORY bsp/ DESTINATION include)
install(TARGETS robohub-board-ky-009-rgb-led-module ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
